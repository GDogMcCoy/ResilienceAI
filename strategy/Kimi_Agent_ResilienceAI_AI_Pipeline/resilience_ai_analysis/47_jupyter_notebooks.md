        template_id: str,
        data: Dict[str, Any],
        output_format: str = 'html',
        output_name: Optional[str] = None
    ) -> str:
        """
        Generate a report from a template.
        
        Args:
            template_id: Template ID
            data: Data to populate template
            output_format: Output format
            output_name: Output filename
            
        Returns:
            Path to generated report
        """
        template = self.templates.get(template_id)
        if not template:
            raise ValueError(f"Template {template_id} not found")
        
        # Generate output filename
        if not output_name:
            timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
            output_name = f"{template.name.lower().replace(' ', '_')}_{timestamp}.{output_format}"
        
        output_path = self.output_dir / output_name
        
        # Render template
        rendered_content = self._render_template(template, data)
        
        # Export to desired format
        if output_format == 'html':
            output_path = self._export_to_html(rendered_content, output_path)
        elif output_format == 'pdf':
            output_path = self._export_to_pdf(rendered_content, output_path)
        elif output_format == 'markdown':
            output_path = self._export_to_markdown(rendered_content, output_path)
        else:
            raise ValueError(f"Unsupported output format: {output_format}")
        
        return str(output_path)
    
    def _render_template(
        self,
        template: ReportTemplate,
        data: Dict[str, Any]
    ) -> str:
        """Render a template with data."""
        # Build report content
        sections_html = []
        
        for section in template.sections:
            section_template = Template(section.content)
            section_content = section_template.render(**data)
            
            sections_html.append(f"""
            <section class="report-section" id="section-{section.order}">
                <h2>{section.title}</h2>
                {section_content}
            </section>
            """)
        
        # Build table of contents
        toc_items = [
            f'<li><a href="#section-{s.order}">{s.title}</a></li>'
            for s in template.sections if s.include_in_toc
        ]
        
        toc_html = f"""
        <nav class="report-toc">
            <h2>Table of Contents</h2>
            <ul>{''.join(toc_items)}</ul>
        </nav>
        """
        
        # Combine everything
        full_report = f"""
        <!DOCTYPE html>
        <html>
        <head>
            <title>{template.name}</title>
            <style>
                {self._get_default_styles()}
                {template.style.get('custom_css', '')}
            </style>
        </head>
        <body>
            <header class="report-header">

            <footer class="report-footer">
                <p>Generated by ResilienceAI Automated Reporting</p>
            </footer>
        </body>
        </html>
        """
        
        return full_report

    def _export_to_html(self, content: str, output_path: Path) -> Path:
        """Export content to HTML file."""
        output_path = output_path.with_suffix('.html')
        with open(output_path, 'w', encoding='utf-8') as f:
            f.write(content)
        return output_path
    
    def _export_to_pdf(self, content: str, output_path: Path) -> Path:
        """Export content to PDF file."""
        # Use weasyprint or similar for PDF conversion
        try:
            from weasyprint import HTML
            output_path = output_path.with_suffix('.pdf')
            HTML(string=content).write_pdf(str(output_path))
            return output_path
        except ImportError:
            # Fallback: save as HTML with PDF extension
            output_path = output_path.with_suffix('.html')
            with open(output_path, 'w', encoding='utf-8') as f:
                f.write(content)
            return output_path
    
    def _export_to_markdown(self, content: str, output_path: Path) -> Path:
        """Export content to Markdown file."""
        # Simple HTML to Markdown conversion
        from html.parser import HTMLParser
        
        class HTMLToMarkdown(HTMLParser):
            def __init__(self):
                super().__init__()
                self.markdown = []
                self.in_script = False
            
            def handle_starttag(self, tag, attrs):
                if tag in ['script', 'style']:
                    self.in_script = True
                elif tag == 'h1':
                    self.markdown.append('\n# ')
                elif tag == 'h2':
                    self.markdown.append('\n## ')
                elif tag == 'p':
                    self.markdown.append('\n')
                elif tag == 'br':
                    self.markdown.append('\n')
                elif tag == 'strong' or tag == 'b':
                    self.markdown.append('**')
                elif tag == 'em' or tag == 'i':
                    self.markdown.append('*')
                elif tag == 'a':
                    for attr, value in attrs:
                        if attr == 'href':
                            self.markdown.append(f'[{value}](')
                elif tag == 'img':
                    src = dict(attrs).get('src', '')
                    alt = dict(attrs).get('alt', '')
                    self.markdown.append(f'![{alt}]({src})')
            
            def handle_endtag(self, tag):
                if tag in ['script', 'style']:
                    self.in_script = False
                elif tag in ['h1', 'h2', 'p']:
                    self.markdown.append('\n')
                elif tag in ['strong', 'b']:
                    self.markdown.append('**')
                elif tag in ['em', 'i']:
                    self.markdown.append('*')
                elif tag == 'a':
                    self.markdown.append(')')
            
            def handle_data(self, data):
                if not self.in_script:
                    self.markdown.append(data)
        
        converter = HTMLToMarkdown()
        converter.feed(content)
        
        output_path = output_path.with_suffix('.md')
        with open(output_path, 'w', encoding='utf-8') as f:
            f.write(''.join(converter.markdown))
        
        return output_path
    
    def schedule_report(
        self,
        name: str,
        template_id: str,
        data_sources: List[str],
        schedule: str,
        recipients: List[str],
        output_format: str = 'html'
    ) -> str:
        """
        Schedule an automated report.
        
        Args:
            name: Report name
            template_id: Template ID
            data_sources: Data source identifiers
            schedule: Cron expression for scheduling
            recipients: Email recipients
            output_format: Output format
            
        Returns:
            Scheduled report ID
        """
        import uuid
        
        scheduled_report = ScheduledReport(
            id=str(uuid.uuid4()),
            name=name,
            template_id=template_id,
            data_sources=data_sources,
            schedule=schedule,
            recipients=recipients,
            format=output_format
        )
        
        self.scheduled_reports[scheduled_report.id] = scheduled_report
        
        # Register with scheduler
        from .scheduler import get_scheduler
        scheduler = get_scheduler()
        
        scheduler.add_task(
            name=f"report_{scheduled_report.id}",
            notebook_path="/mnt/okcomputer/notebooks/05_reports/01_automated_reports.ipynb",
            schedule_type='cron',
            schedule_config={'at': schedule},
            parameters={
                'template_id': template_id,
                'data_sources': data_sources,
                'recipients': recipients,
                'output_format': output_format
            }
        )
        
        return scheduled_report.id
